**SeatCheckerRealtime.tsx** — "use client" component:

This component wraps or enhances the existing seat checker in the WaitlistModal to receive real-time updates when admin changes zip_seats data.

First, examine how the WaitlistModal currently fetches seat data. It likely calls `/api/reservation/seats?zip=X&role=Y`. The Realtime component needs to:

1. Subscribe to `postgres_changes` on `zip_seats` table for UPDATE events
2. When a zip_seats UPDATE event arrives, if the updated row matches the currently-viewed ZIP and role in the modal, refetch the seat data from `/api/reservation/seats` and update the displayed count

Implementation approach:

```typescript
"use client"
import { useEffect, useCallback } from "react"
import { createClient } from "@/lib/supabase/client"

interface SeatCheckerRealtimeProps {
  zipCode: string | null
  role: string | null
  onSeatUpdate: (data: { seatsRemaining: number; totalCap: number }) => void
}

export function SeatCheckerRealtime({ zipCode, role, onSeatUpdate }: SeatCheckerRealtimeProps) {
  const refetchSeats = useCallback(async () => {
    if (!zipCode || !role) return
    try {
      const res = await fetch(`/api/reservation/seats?zip=${zipCode}&role=${role}`)
      const data = await res.json()
      if (data.inArea) {
        onSeatUpdate({ seatsRemaining: data.seatsRemaining, totalCap: data.totalCap })
      }
    } catch {
      // Non-critical — seat display will be stale until next manual check
    }
  }, [zipCode, role, onSeatUpdate])

  useEffect(() => {
    if (!zipCode) return

    const supabase = createClient()
    const channel = supabase
      .channel(`seat-checker-${zipCode}`)
      .on(
        "postgres_changes",
        {
          event: "UPDATE",
          schema: "public",
          table: "zip_seats",
          filter: `zip_code=eq.${zipCode}`,
        },
        () => {
          // Refetch fresh seat data on any zip_seats update for this ZIP
          refetchSeats()
        }
      )
      .subscribe()

    return () => {
      supabase.removeChannel(channel)
    }
  }, [zipCode, refetchSeats])

  return null // Renders nothing — purely a side-effect component
}
```

This component renders nothing — it just subscribes and calls `onSeatUpdate` when data changes. The WaitlistModal or its parent can include this component and pass the callback to update its seat display state.

IMPORTANT: This component is additive. It does NOT modify the WaitlistModal itself — it provides a Realtime hook that can be composed alongside the modal. The WaitlistModal already fetches seat data on ZIP input; this component adds live updates after the initial fetch.

If the WaitlistModal manages seat state in its useReducer, the parent (page.tsx or the modal itself) can wire the `onSeatUpdate` callback to dispatch a state update.

Document in the component's JSDoc how to wire it:
```
// Usage: Place <SeatCheckerRealtime zipCode={selectedZip} role={selectedRole} onSeatUpdate={handleUpdate} />
// alongside the WaitlistModal. When admin updates zip_seats, this fires onSeatUpdate with fresh data.
```

Note: The Realtime publication `ALTER PUBLICATION supabase_realtime ADD TABLE zip_seats;` is applied by Plan 01 Task 1's migration. Without that migration, this subscription will silently receive no events.
